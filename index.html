Microsoft Sentinel KQL Rule (Impossible Travel)

| summarize FirstSeen = min(TimeGenerated), LastSeen = max(TimeGenerated) 
    by UserPrincipalName, IPAddress, Location
| extend PrevLocation = prev(Location), PrevTime = prev(LastSeen)
| where Location != PrevLocation and datetime_diff('minute', FirstSeen, PrevTime) < 60
| project UserPrincipalName, Location, PrevLocation, IPAddress, FirstSeen
